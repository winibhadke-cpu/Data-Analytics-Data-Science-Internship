Total Revenue = SUM(Fact_Sales[Sales])
Total Orders = DISTINCTCOUNT(Fact_Sales[Order_ID])
Total Customers = DISTINCTCOUNT(Fact_Sales[Customer_Key])
Average Order Value = DIVIDE([Total Revenue], [Total Orders])
Total Units Sold = SUM(Fact_Sales[Quantity])
Total Profit = BLANK()
Profit Margin = DIVIDE([Total Profit], [Total Revenue])
Previous Month Revenue = CALCULATE([Total Revenue], DATEADD(Dim_Date[Date], -1, MONTH))
Monthly Growth = IF(HASONEVALUE(Dim_Date[Year]) && HASONEVALUE(Dim_Date[Month_Number]) && NOT ISBLANK([Total Revenue]) && NOT ISBLANK([Previous Month Revenue]), DIVIDE([Total Revenue] - [Previous Month Revenue], [Previous Month Revenue]), BLANK())

Total Products = DISTINCTCOUNT(Fact_Sales[Product_Key])
Average Customer Spending = DIVIDE([Total Revenue], [Total Customers])
Repeat Customers = COUNTROWS(FILTER(VALUES(Fact_Sales[Customer_Key]), CALCULATE([Total Orders]) > 1))
Repeat Customer Rate = DIVIDE([Repeat Customers], [Total Customers])
Customer Rank = IF(NOT ISBLANK([Total Revenue]), RANKX(ALLSELECTED(Dim_Customer[Customer_Name]), [Total Revenue], , DESC, DENSE))
Product Rank = IF(NOT ISBLANK([Total Revenue]), RANKX(ALLSELECTED(Dim_Product[Product_Name]), [Total Revenue], , DESC, DENSE))
Display Product Profit = IF(ISBLANK([Total Profit]), "N/A", FORMAT([Total Profit], "#,##0.00"))
